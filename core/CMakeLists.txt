cmake_minimum_required(VERSION 3.16)
project(solar_core VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(CURL REQUIRED)
find_package(ZLIB REQUIRED)
find_package(Threads REQUIRED)
add_library(solar_logic STATIC src/auth.cpp src/platform.cpp src/game.cpp src/archive.cpp src/runtime.cpp src/downloads.cpp src/resources.cpp src/mods.cpp src/packs.cpp src/modinstall.cpp src/javaprovider.cpp src/gameinstall.cpp src/loaderinstall.cpp src/optifine.cpp src/engine.cpp src/accounts.cpp src/credentialvault.cpp third_party/tinyxml2.cpp)
target_include_directories(solar_logic PUBLIC include third_party)
target_link_libraries(solar_logic PUBLIC CURL::libcurl ZLIB::ZLIB Threads::Threads)
if(APPLE)
 target_link_libraries(solar_logic PUBLIC "-framework Security")
elseif(WIN32)
 target_compile_definitions(solar_logic PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0601)
 target_link_libraries(solar_logic PUBLIC bcrypt shell32 ws2_32 advapi32)
 target_compile_options(solar_logic PUBLIC /utf-8)
else()
 find_package(PkgConfig REQUIRED)
 pkg_check_modules(SECRET REQUIRED IMPORTED_TARGET libsecret-1)
 target_link_libraries(solar_logic PUBLIC PkgConfig::SECRET)
 find_package(OpenSSL REQUIRED)
 target_link_libraries(solar_logic PUBLIC OpenSSL::Crypto)
endif()
add_library(solar_core SHARED src/api.cpp)
target_link_libraries(solar_core PRIVATE solar_logic)
set_target_properties(solar_core PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(APPLE)
 set_target_properties(solar_core PROPERTIES INSTALL_NAME_DIR "@rpath")
endif()
include(CTest)
if(BUILD_TESTING)
 add_executable(solar_core_tests tests/core_tests.cpp)
 target_link_libraries(solar_core_tests PRIVATE solar_logic)
 add_test(NAME solar_core_tests COMMAND solar_core_tests)
 add_executable(solar_engine_tests tests/engine_tests.cpp)
 target_link_libraries(solar_engine_tests PRIVATE solar_logic)
 add_test(NAME solar_engine_tests COMMAND solar_engine_tests)
endif()
